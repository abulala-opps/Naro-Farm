    <!-- ========== 评论区 JS（假评论区，刷新就丢） ========== -->
    <script>
        const nameInput = document.getElementById('commentName');
        const bodyInput = document.getElementById('commentBody');
        const submitBtn = document.getElementById('commentSubmit');
        const list = document.getElementById('commentList');

        // 提交留言
        submitBtn.addEventListener('click', function() {
            const name = nameInput.value.trim() || '匿名访客';
            const body = bodyInput.value.trim();

            if (!body) {
                alert('留言内容不能为空哦～');
                return;
            }

            // 移除"还没有留言"提示
            const empty = list.querySelector('.empty-tip');
            if (empty) empty.remove();

            // 创建留言元素
            const item = document.createElement('div');
            item.className = 'comment-item';

            const now = new Date();
            const timeStr = now.getFullYear() + '/' +
                            (now.getMonth() + 1) + '/' +
                            now.getDate() + ' ' +
                            String(now.getHours()).padStart(2, '0') + ':' +
                            String(now.getMinutes()).padStart(2, '0');

            item.innerHTML = `
                <div class="comment-head">
                    <span class="comment-name">${escapeHtml(name)}</span>
                    <span class="comment-time">${timeStr}</span>
                </div>
                <div class="comment-body">${escapeHtml(body)}</div>
            `;

            // 插入到最上面
            list.insertBefore(item, list.firstChild);

            // 清空输入框
            nameInput.value = '';
            bodyInput.value = '';
        });

        // 简单的 HTML 转义，防止用户输入标签
        function escapeHtml(str) {
            return str.replace(/[&<>"']/g, function(m) {
                return {
                    '&': '&amp;',
                    '<': '&lt;',
                    '>': '&gt;',
                    '"': '&quot;',
                    "'": '&#39;'
                }[m];
            });
        }
    </script>
